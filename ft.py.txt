import streamlit as st
import pandas as pd
import io
import xml.etree.ElementTree as ET

def efatura_mal_iceriklerini_oku(uploaded_files):
    """
    Streamlit üzerinden gelen UBL-TR E-Fatura XML dosyalarýndan 
    mal/hizmet satýrlarýný (InvoiceLine) ayrýþtýrarak tek bir DataFrame toplar.
    """
    tum_satirlar = []
    
    for uf in uploaded_files:
        try:
            tree = ET.parse(uf)
            root = tree.getroot()
            
            # Fatura No ve Tarih bilgilerini yakala
            fatura_no, fatura_tarihi, firma_unvani = "Bilinmiyor", "", "Bilinmiyor"
            for elem in root.iter():
                tag = elem.tag.split('}')[-1].lower()
                text = elem.text.strip() if elem.text else ""
                if tag in ["id", "faturano"] and fatura_no == "Bilinmiyor" and text:
                    fatura_no = text
                elif tag in ["issuedate", "tarih"] and not fatura_tarihi and text:
                    fatura_tarihi = text
                elif tag in ["partyname", "name", "unvan"] and firma_unvani == "Bilinmiyor" and text:
                    firma_unvani = text

            # Mal ve hizmet kalemlerini (Line / InvoiceLine) tara
            for line in root.iter():
                tag = line.tag.split('}')[-1].lower()
                if tag in ["invoiceline", "line"]:
                    mal_adi = "Tanýmsýz Mal / Hizmet"
                    miktar = 1.0
                    birim_fiyat = 0.0
                    tutar = 0.0
                    kdv_orani = 20.0
                    
                    for child in line.iter():
                        ctag = child.tag.split('}')[-1].lower()
                        cval = child.text.strip() if child.text else ""
                        
                        if ctag in ["name", "itemname", "description"] and cval and mal_adi == "Tanýmsýz Mal / Hizmet":
                            mal_adi = cval
                        elif ctag in ["invoicedquantity", "quantity"]:
                            try: miktar = float(cval)
                            except: pass
                        elif ctag in ["priceamount", "price"]:
                            try: birim_fiyat = float(cval)
                            except: pass
                        elif ctag in ["lineextensionamount", "amount"]:
                            try: tutar = float(cval)
                            except: pass
                        elif ctag in ["percent"] and cval in ["1", "8", "10", "18", "20"]:
                            try: kdv_orani = float(cval)
                            except: pass
                    
                    if birim_fiyat == 0.0 and miktar > 0 and tutar > 0:
                        birim_fiyat = tutar / miktar
                    elif tutar == 0.0 and miktar > 0 and birim_fiyat > 0:
                        tutar = miktar * birim_fiyat

                    tum_satirlar.append({
                        "Fatura No": fatura_no,
                        "Tarih": fatura_tarihi,
                        "Firma Unvaný": firma_unvani,
                        "Mal / Hizmet Açýklamasý": mal_adi,
                        "Miktar": miktar,
                        "Birim Fiyat (TL)": birim_fiyat,
                        "Tutar (TL)": tutar,
                        "KDV Oraný (%)": kdv_orani
                    })
        except Exception as e:
            st.error(f"Dosya okuma hatasý ({uf.name}): {e}")
            
    return pd.DataFrame(tum_satirlar)
